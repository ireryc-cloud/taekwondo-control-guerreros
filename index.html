    <script>

        // Arreglo de cintas en orden exacto

        const ordenCintas = [

            "Blanca", "Blanca Avanzada", "Naranja", "Amarilla", "Amarilla Avanzada",

            "Verde", "Verde Avanzada", "Azul", "Azul Avanzada", "Roja",

            "1er iebi", "2do iebi", "3er iebi", "Negra"

        ];

 

        // Base de Datos inicial por defecto si está vacía

        let db = JSON.parse(localStorage.getItem('guerreros_db')) || {

            maestros: [

                { id: "m1", nombre: "Prof. Carlos", clases: 12 }

            ],

            alumnos: [

                { id: "a1", nombre: "Santiago Ramírez", cinta: "Amarilla", fecha_examen: "2026-08-15", asistencias: 24, monto: 500, estado_pago: "Pagado", tel: "5512345678" },

                { id: "a2", nombre: "Sofía Martínez", cinta: "Naranja", fecha_examen: "2026-09-01", asistencias: 28, monto: 600, estado_pago: "Pendiente", tel: "5587654321" }

            ],

            movimientos: [

                { id: "mov1", fecha: "2026-11-02", tipo: "ingreso", concepto: "Venta de Dobok", monto: 850 },

                { id: "mov2", fecha: "2026-11-05", tipo: "egreso", concepto: "Renta del Dojang", monto: 4500 }

            ]

        };

 

        function guardarDB() {

            localStorage.setItem('guerreros_db', JSON.stringify(db));

            renderTodo();

        }

 

        // Navegación entre pestañas

        function cambiarTab(tab) {

            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));

            document.querySelectorAll('.tab-btn').forEach(el => {

                el.classList.remove('border-red-900', 'font-bold', 'text-red-900');

                el.classList.add('border-transparent', 'text-gray-600');

            });

 

            document.getElementById(`sec-${tab}`).classList.remove('hidden');

            const btn = document.getElementById(`btn-${tab}`);

            btn.classList.add('border-red-900', 'font-bold', 'text-red-900');

            btn.classList.remove('border-transparent', 'text-gray-600');

        }

 

        // Renderizado general

        function renderTodo() {

            renderAlumnos();

            renderCintas();

            renderMaestros();

            renderFinanzas();

        }

 

        // 1. Render Alumnos & Asistencia

        function renderAlumnos() {

            // Ordenar alumnos por asistencias de mayor a menor (Ranking automático)

            db.alumnos.sort((a, b) => b.asistencias - a.asistencias);

            const tbody = document.getElementById('tabla-alumnos-body');

            tbody.innerHTML = '';

 

            db.alumnos.forEach((al, index) => {

                let tr = document.createElement('tr');

                tr.className = index % 2 === 0 ? 'bg-white hover:bg-gray-50' : 'bg-gray-50 hover:bg-gray-100';

               

                let badgePago = al.estado_pago === 'Pagado'

                    ? `<span class="bg-green-100 text-green-800 px-2 py-1 rounded text-xs font-bold cursor-pointer" onclick="togglePago('${al.id}')">Pagado</span>`

                    : `<span class="bg-red-100 text-red-800 px-2 py-1 rounded text-xs font-bold cursor-pointer" onclick="togglePago('${al.id}')">Pendiente</span>`;

 

                let btnWsp = `<a href="https://wa.me/52${al.tel}?text=Hola,%20te%20saludamos%20de%20Guerreros%20Aztecas.%20Te%20recordamos%20que%20tu%20mensualidad%20de%20$%${al.monto}%20está%20pendiente.%20¡Saludos!" target="_blank" class="bg-green-600 hover:bg-green-700 text-white px-2 py-1 rounded text-xs inline-flex items-center ml-1"><i class="fa-brands fa-whatsapp mr-1"></i> Cobrar</a>`;

 

                tr.innerHTML = `

                    <td class="p-3 font-bold text-gray-700">#${index + 1}</td>

                    <td class="p-3 font-medium text-gray-900">${al.nombre}</td>

                    <td class="p-3 text-gray-600">${al.cinta}</td>

                    <td class="p-3 text-center">

                        <span class="font-bold text-lg text-red-900">${al.asistencias}</span>

                        <button onclick="sumarAsistencia('${al.id}')" class="ml-2 bg-blue-600 hover:bg-blue-700 text-white px-2 py-1 rounded text-xs"><i class="fa-solid fa-plus"></i></button>

                    </td>

                    <td class="p-3"><input type="number" value="${al.monto}" onchange="actualizarMonto('${al.id}', this.value)" class="w-24 border p-1 rounded text-sm"></td>

                    <td class="p-3">${badgePago}</td>

                    <td class="p-3 text-center">

                        ${al.estado_pago === 'Pendiente' ? btnWsp : ''}

                        <button onclick="eliminarAlumno('${al.id}')" class="text-red-500 hover:text-red-700 ml-2"><i class="fa-solid fa-trash"></i></button>

                    </td>

                `;

                tbody.appendChild(tr);

            });

        }

 

        function sumarAsistencia(id) {

            let al = db.alumnos.find(x => x.id === id);

            if(al) {

                al.asistencias += 1;

                guardarDB();

            }

        }

 

        function togglePago(id) {

            let al = db.alumnos.find(x => x.id === id);

            if(al) {

                al.estado_pago = al.estado_pago === 'Pagado' ? 'Pendiente' : 'Pagado';

                guardarDB();

            }

        }

 

        function actualizarMonto(id, nuevoMonto) {

            let al = db.alumnos.find(x => x.id === id);

            if(al) {

                al.monto = parseFloat(nuevoMonto) || 0;

                guardarDB();

            }

        }

 

        // 2. Render Cintas & Evaluaciones (Ciclo de 3 meses)

        function renderCintas() {

            const tbody = document.getElementById('tabla-cintas-body');

            tbody.innerHTML = '';

 

            db.alumnos.forEach(al => {

                let fechaUlt = new Date(al.fecha_examen);

                let fechaProx = new Date(fechaUlt);

                fechaProx.setMonth(fechaProx.getMonth() + 3); // Sumar 3 meses

 

                let hoy = new Date();

                let diffDias = Math.ceil((fechaProx - hoy) / (1000 * 60 * 60 * 24));

 

                let estadoSem = "";

                if(diffDias <= 0) {

                    estadoSem = `<span class="bg-green-100 text-green-800 px-2 py-1 rounded text-xs font-bold">🟢 Listo para Examen</span>`;

                } else if(diffDias <= 15) {

                    estadoSem = `<span class="bg-yellow-100 text-yellow-800 px-2 py-1 rounded text-xs font-bold">🟡 Próximo (${diffDias} días)</span>`;

                } else {

                    estadoSem = `<span class="bg-gray-100 text-gray-800 px-2 py-1 rounded text-xs">En preparación</span>`;

                }

 

                // Identificar siguiente cinta

                let idxActual = ordenCintas.indexOf(al.cinta);

                let siguienteCinta = (idxActual !== -1 && idxActual < ordenCintas.length - 1) ? ordenCintas[idxActual + 1] : "Grado Máximo";

 

                let tr = document.createElement('tr');

                tr.className = "hover:bg-gray-50";

                tr.innerHTML = `

                    <td class="p-3 font-medium text-gray-900">${al.nombre}</td>

                    <td class="p-3 text-gray-700 font-semibold">${al.cinta}</td>

                    <td class="p-3 text-gray-600">${al.fecha_examen}</td>

                    <td class="p-3 text-gray-600">${fechaProx.toISOString().split('T')[0]}</td>

                    <td class="p-3 text-amber-700 font-bold">${siguienteCinta}</td>

                    <td class="p-3">${estadoSem}</td>

                    <td class="p-3 text-center">

                        <button onclick="promoverAlumno('${al.id}')" class="bg-green-600 hover:bg-green-700 text-white px-3 py-1 rounded text-xs font-semibold"><i class="fa-solid fa-check mr-1"></i> Aprobar / Subir Grado</button>

                    </td>

                `;

                tbody.appendChild(tr);

            });

        }

 

        function promoverAlumno(id) {

            let al = db.alumnos.find(x => x.id === id);

            if(al) {

                let idxActual = ordenCintas.indexOf(al.cinta);

                if(idxActual < ordenCintas.length - 1) {

                    al.cinta = ordenCintas[idxActual + 1];

                    al.fecha_examen = new Date().toISOString().split('T')[0]; // Reinicia reloj de 3 meses a hoy

                    guardarDB();

                    alert(`¡Felicidades a ${al.nombre}! Ha ascendido a cinta ${al.cinta}`);

                } else {

                    alert("El alumno ya se encuentra en el grado máximo.");

                }

            }

        }

 

        // 3. Render Maestros

        function renderMaestros() {

            const grid = document.getElementById('grid-maestros');

            grid.innerHTML = '';

 

            db.maestros.forEach(m => {

                let card = document.createElement('div');

                card.className = "bg-white p-4 rounded-lg shadow border border-gray-200 flex flex-col justify-between";

                card.innerHTML = `

                    <div>

                        <h3 class="font-bold text-lg text-gray-800">${m.nombre}</h3>

                        <p class="text-sm text-gray-600 mt-2">Clases impartidas: <span class="text-2xl font-bold text-red-900">${m.clases}</span></p>

                    </div>

                    <div class="mt-4 flex justify-between items-center">

                        <button onclick="sumarClaseMaestro('${m.id}')" class="bg-blue-600 hover:bg-blue-700 text-white px-3 py-2 rounded text-sm font-semibold w-full mr-2"><i class="fa-solid fa-plus mr-1"></i> +1 Clase</button>

                        <button onclick="eliminarMaestro('${m.id}')" class="text-red-500 hover:text-red-700 p-2"><i class="fa-solid fa-trash"></i></button>

                    </div>

                `;

                grid.appendChild(card);

            });

        }

 

        function sumarClaseMaestro(id) {

            let m = db.maestros.find(x => x.id === id);

            if(m) {

                m.clases += 1;

                guardarDB();

            }

        }

 

        // 4. Finanzas & Balance

        function renderFinanzas() {

            // Ingresos automáticos por mensualidades pagadas de alumnos

            let ingresosMensualidades = db.alumnos

                .filter(a => a.estado_pago === 'Pagado')

                .reduce((sum, a) => sum + (parseFloat(a.monto) || 0), 0);

 

            let ingresosExtras = db.movimientos

                .filter(m => m.tipo === 'ingreso')

                .reduce((sum, m) => sum + (parseFloat(m.monto) || 0), 0);

 

            let totalIngresos = ingresosMensualidades + ingresosExtras;

 

            let totalEgresos = db.movimientos

                .filter(m => m.tipo === 'egreso')

                .reduce((sum, m) => sum + (parseFloat(m.monto) || 0), 0);

 

            let balanceNeto = totalIngresos - totalEgresos;

 

            document.getElementById('lbl-total-ingresos').innerText = `$${totalIngresos.toLocaleString('es-MX')}`;

            document.getElementById('lbl-total-egresos').innerText = `$${totalEgresos.toLocaleString('es-MX')}`;

            document.getElementById('lbl-balance-neto').innerText = `$${balanceNeto.toLocaleString('es-MX')}`;

 

            // Tabla de movimientos

            const tbody = document.getElementById('tabla-movimientos-body');

            tbody.innerHTML = `

                <tr class="bg-green-50">

                    <td class="p-3 text-gray-600">Mes Actual</td>

                    <td class="p-3"><span class="bg-green-200 text-green-900 px-2 py-0.5 rounded text-xs font-bold">Ingreso Auto</span></td>

                    <td class="p-3 font-medium">Mensualidades de Alumnos Pagados</td>

                    <td class="p-3 text-green-700 font-bold">+$${ingresosMensualidades.toLocaleString('es-MX')}</td>

                    <td class="p-3"></td>

                </tr>

            `;

 

            db.movimientos.forEach(mov => {

                let tr = document.createElement('tr');

                tr.className = "hover:bg-gray-50";

                let badge = mov.tipo === 'ingreso'

                    ? `<span class="bg-green-100 text-green-800 px-2 py-0.5 rounded text-xs font-bold">Ingreso</span>`

                    : `<span class="bg-red-100 text-red-800 px-2 py-0.5 rounded text-xs font-bold">Egreso</span>`;

               

                tr.innerHTML = `

                    <td class="p-3 text-gray-600">${mov.fecha}</td>

                    <td class="p-3">${badge}</td>

                    <td class="p-3 font-medium text-gray-800">${mov.concepto}</td>

                    <td class="p-3 font-bold ${mov.tipo === 'ingreso' ? 'text-green-700' : 'text-red-700'}">$${parseFloat(mov.monto).toLocaleString('es-MX')}</td>

                    <td class="p-3 text-center"><button onclick="eliminarMovimiento('${mov.id}')" class="text-red-500 hover:text-red-700"><i class="fa-solid fa-trash"></i></button></td>

                `;

                tbody.appendChild(tr);

            });

 

            // Fecha por defecto hoy en el formulario

            document.getElementById('mov-fecha').value = new Date().toISOString().split('T')[0];

        }

 

        function agregarMovimiento(e) {

            e.preventDefault();

            let nuevoMov = {

                id: 'mov_' + Date.now(),

                tipo: document.getElementById('mov-tipo').value,

                fecha: document.getElementById('mov-fecha').value,

                concepto: document.getElementById('mov-concepto').value,

                monto: parseFloat(document.getElementById('mov-monto').value)

            };

            db.movimientos.push(nuevoMov);

            guardarDB();

            e.target.reset();

            document.getElementById('mov-fecha').value = new Date().toISOString().split('T')[0];

        }

 

        function eliminarMovimiento(id) {

            db.movimientos = db.movimientos.filter(m => m.id !== id);

            guardarDB();

        }

        // Exportar / Importar respaldo JSON

        function exportarJSON() {

            let dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(db, null, 2));

            let downloadAnchor = document.createElement('a');

            downloadAnchor.setAttribute("href", dataStr);

            downloadAnchor.setAttribute("download", `guerreros_aztecas_respaldo_${new Date().toISOString().split('T')[0]}.json`);

            document.body.appendChild(downloadAnchor);

            downloadAnchor.click();

            downloadAnchor.remove();

        }

 

        function importarJSON(event) {

            let file = event.target.files[0];

            if (!file) return;

            let reader = new FileReader();

            reader.onload = function(e) {

                try {

                    db = JSON.parse(e.target.result);

                    guardarDB();

                    alert("¡Respaldo cargado correctamente!");

                } catch (err) {

                    alert("El archivo JSON no es válido.");

                }

            };

            reader.readAsText(file);

        }
